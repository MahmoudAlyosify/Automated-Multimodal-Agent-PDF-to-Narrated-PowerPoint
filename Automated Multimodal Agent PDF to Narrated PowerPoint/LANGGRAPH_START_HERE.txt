STREAMLIT INTEGRATION
═══════════════════════════════════════════════════════════════════════════

import streamlit as st
from langgraph_orchestrator import run_orchestration

uploaded = st.file_uploader("Upload PDF", type="pdf")
if uploaded:
    with open("temp.pdf", "wb") as f:
        f.write(uploaded.getvalue())
    
    if st.button("Convert to PowerPoint"):
        with st.spinner("Processing..."):
            result = run_orchestration("temp.pdf", "output.pptx")
        
        if result['status'] == 'completed':
            st.success("✓ Conversion complete!")
            with open(result['final_output'], 'rb') as f:
                st.download_button(
                    "Download PowerPoint",
                    f,
                    file_name="presentation.pptx"
                )
        else:
            st.error(f"Error: {result['errors'][0]}")

  STATE STRUCTURE (WHAT FLOWS THROUGH THE GRAPH)
█████████████████████████████████████████████████████████████████████████████

OrchestrationState = {
    # INPUT PARAMETERS
    "pdf_path": str,                    # Input PDF file
    "output_pptx": str,                 # Output PowerPoint
    "start_page": Optional[int],        # Page range start
    "end_page": Optional[int],          # Page range end
    "domain": str,                      # Document domain
    "language": str,                    # Language code
    "venv_path": Optional[str],         # Virtual environment
    
    # STEP 1: DOCUMENT UNDERSTANDING RESULTS
    "document_extracted": bool,         # Success flag
    "extracted_content": Optional[Dict],  # Extracted JSON
    "document_error": Optional[str],    # Error if failed
    
    # STEP 2: BRAIN AGENT RESULTS
    "slides_designed": bool,            # Success flag
    "slides_json": Optional[Dict],      # Slide specs
    "brain_error": Optional[str],       # Error if failed
    
    # STEP 3: PPT RENDERING RESULTS
    "ppt_created": bool,                # Success flag
    "ppt_error": Optional[str],         # Error if failed
    "final_output": Optional[str],      # Output path
    
    # WORKFLOW METADATA
    "status": str,                      # Current status
    "errors": list[str]                 # All errors
}
